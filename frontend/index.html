const BASE = '__BASE_URL__';

interface Manga {
  title: string; slug: string;
  thumbnail: string; originalLink: string; apiDetailLink: string;
}
interface SearchResult {
  status: boolean; keyword: string; total: number; data: Manga[];
}

async function cariManga(q: string): Promise<SearchResult> {
  const res = await fetch(`${BASE}/search?q=${encodeURIComponent(q)}`);
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return res.json();
}

async function getDetail(slug: string) {
  const res = await fetch(`${BASE}/detail-komik/${slug}`);
  return res.json();
}

async function getChapter(slug: string, chapter: string) {
  const res = await fetch(`${BASE}/baca-chapter/${slug}/${chapter}`);
  return res.json();
}
import { useEffect, useState } from 'react';

const BASE = '__BASE_URL__';

// Hook: ambil komik terbaru
export function useTerbaru() {
  const [data, setData] = useState([]);
  const [loading, setLoading] = useState(true);
  useEffect(() => {
    fetch(`${BASE}/terbaru`)
      .then(r => r.json())
      .then(d => { setData(d); setLoading(false); });
  }, []);
  return { data, loading };
}

// Hook: cari manga
export function useSearch(query) {
  const [result, setResult] = useState(null);
  useEffect(() => {
    if (!query) return;
    fetch(`${BASE}/search?q=${encodeURIComponent(query)}`)
      .then(r => r.json())
      .then(setResult);
  }, [query]);
  return result;
}

// Komponen contoh
export default function MangaList() {
  const { data, loading } = useTerbaru();
  if (loading) return <p>Loading...</p>;
  return (
    <ul>
      {data.map(m => (
        <li key={m.slug}>
          <img src={`${BASE}/image-proxy?url=${encodeURIComponent(m.thumbnail)}`} alt={m.title} />
          <span>{m.title}</span>
        </li>
      ))}
    </ul>
  );
}

Rate Limit

Fair-use limit agar layanan stabil untuk semua.

⚠ 200 request per 15 menit per IP. Melebihi limit → HTTP 429. Header RateLimit-Limit dan RateLimit-Remaining disertakan di setiap response. 
HTTP/1.1 429 Too Many Requests
RateLimit-Limit: 200
RateLimit-Remaining: 0
RateLimit-Reset: 900

{
  "error": "Terlalu banyak request dari IP ini, silakan coba lagi setelah 15 menit"
}

Image Proxy
GET /image-proxy?url=... Proxy gambar cover / chapter

Mem-proxy gambar cover dan chapter melalui server untuk bypass hotlink protection CDN sumber. Return binary gambar dengan Content-Type dan cache 24 jam.

Query Parameter
NamaTipeWajibKeterangan
url string required URL lengkap gambar yang akan di-proxy (dari CDN komiku)
Contoh Penggunaan
<!-- Di HTML -->
<img src="__BASE_URL__/image-proxy?url=https://thumbnail.komiku.org/uploads/manga/bocchi-the-rock/cover.jpg" />

// Di JavaScript
const imgSrc = `${BASE}/image-proxy?url=${encodeURIComponent(manga.thumbnail)}`;
